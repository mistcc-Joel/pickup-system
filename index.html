<script>

  // ⚠️【非常重要】請將這裡替換成您剛剛部署 GAS 產生的「網頁應用程式網址」

  const GAS_API_URL = "https://script.google.com/macros/s/AKfycbyTbufOyqkkVxTnDbOReNbspnxkUv2chfN9RmmQKF0cEfw3eySM7OEdmV2nn0THsdBMrA/exec";

  

  let currentPin = ""; 

  let html5QrcodeScanner = null;

  let selectedStudent = null;

  const canvas = document.getElementById('signature-pad');

  const ctx = canvas.getContext('2d');

  let isDrawing = false;


  // ================= 核心：API 呼叫中心 =================

  async function apiCall(action, data = {}) {

    try {

      const response = await fetch(GAS_API_URL, {

        method: 'POST',

        headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // 避開瀏覽器 CORS 攔截

        body: JSON.stringify({ action: action, pin: currentPin, ...data })

      });

      const result = await response.json();

      if (!result.success && result.message.includes("PIN")) {

        alert("🔒 登入逾時或密碼錯誤，請重新登入！");

        location.reload();

      }

      return result;

    } catch (err) {

      alert("連線到資料庫失敗，請檢查網路連線。");

      return { success: false, message: "連線失敗" };

    }

  }


  // ================= 登入驗證 =================

  async function verifyPin() {

    const pin = document.getElementById('pin-input').value;

    if(!pin) return;

    document.getElementById('login-msg').innerText = "驗證中，請稍候...";

    

    currentPin = pin; // 暫存密碼

    const res = await apiCall('verifyPin');

    

    if (res.success) {

      document.getElementById('login-screen').style.display = "none";

      document.getElementById('app-container').style.display = "block";

    } else {

      document.getElementById('login-msg').innerText = res.message;

      currentPin = "";

    }

  }


  // ================= 頁面切換 =================

  function switchTab(tab) {

    stopScanner();

    ['sign', 'list', 'report'].forEach(t => {

      document.getElementById('tab-' + t).classList.remove('active');

      document.getElementById('section-' + t).classList.remove('active');

    });

    document.getElementById('tab-' + tab).classList.add('active');

    document.getElementById('section-' + tab).classList.add('active');

  }


  // ================= 獨立版相機掃描 (保證不破圖/不擋權限) =================

  function startScanner() {

    document.getElementById('start-scan-btn').style.display = 'none';

    document.getElementById('qr-reader-container').style.display = 'block';

    

    if (!html5QrcodeScanner) {

      html5QrcodeScanner = new Html5QrcodeScanner("reader", { fps: 15, qrbox: { width: 250, height: 250 } }, false);

      html5QrcodeScanner.render(onScanSuccess, (err)=>{});

    }

  }


  function stopScanner() {

    if (html5QrcodeScanner) {

      html5QrcodeScanner.clear();

      html5QrcodeScanner = null;

    }

    document.getElementById('qr-reader-container').style.display = 'none';

    document.getElementById('start-scan-btn').style.display = 'block';

  }


  async function onScanSuccess(decodedText) {

    stopScanner();

    document.getElementById('search-results').innerHTML = "<div style='color:#0284c7; text-align:center;'>🔄 掃描成功！資料比對中...</div>";

    try {

      const qrData = JSON.parse(decodedText);

      if (qrData._membrId) {

        const res = await apiCall('searchMembrId', { membrId: qrData._membrId });

        showResults(res.data);

      }

    } catch(e) { alert("⚠️ 無效的 QR Code 格式"); document.getElementById('search-results').innerHTML = ""; }

  }


  // ================= 查詢與結果顯示 =================

  async function search() {

    const keyword = document.getElementById('keyword').value.trim();

    if (!keyword) return alert("請輸入關鍵字");

    document.getElementById('keyword').blur();

    document.getElementById('search-results').innerHTML = "<div style='color:#94a3b8; text-align:center;'>🔍 查詢中...</div>";

    

    const res = await apiCall('search', { keyword: keyword });

    if(res.success) showResults(res.data);

  }


  function showResults(results) {

    const container = document.getElementById('search-results');

    if (!results || results.length === 0) return container.innerHTML = "<div style='color:#e11d48; text-align:center;'>找不到符合的學童</div>";

    

    if (results.length === 1 && document.getElementById('keyword').value === "") {

      return selectStudent(JSON.stringify(results[0]));

    }

    

    let html = "";

    results.forEach(s => {

      html += `<div class="result-card" onclick="selectStudent('${JSON.stringify(s).replace(/"/g, '&quot;')}')">

                 <div>${s.className} - <span style="color:#0284c7; font-size:22px;">${s.name}</span><div class="result-sub">學號: ${s.studentId}</div></div>

                 <div>❯</div></div>`;

    });

    container.innerHTML = html;

  }


  // ================= 畫布簽名與提交 =================

  function selectStudent(studentJson) {

    selectedStudent = JSON.parse(studentJson);

    document.getElementById('step-search').style.display = "none";

    document.getElementById('step-signature').style.display = "block";

    document.getElementById('display-name').innerText = `${selectedStudent.className} - ${selectedStudent.name}`;

    

    const now = new Date();

    document.getElementById('display-time').innerText = `時間：${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

    setTimeout(resizeCanvas, 100); // 確保畫面渲染後再抓尺寸

  }


  function backToSearch() {

    selectedStudent = null;

    document.getElementById('step-search').style.display = "block";

    document.getElementById('step-signature').style.display = "none";

    document.getElementById('keyword').value = "";

    document.getElementById('search-results').innerHTML = "";

  }


  function resizeCanvas() {

    const ratio = Math.max(window.devicePixelRatio || 1, 1);

    canvas.width = canvas.offsetWidth * ratio;

    canvas.height = canvas.offsetHeight * ratio;

    ctx.scale(ratio, ratio);

    clearCanvas();

  }

  function clearCanvas() { ctx.clearRect(0, 0, canvas.width, canvas.height); }

  function getPos(e) {

    const rect = canvas.getBoundingClientRect();

    const clientX = e.clientX || (e.touches && e.touches[0].clientX);

    const clientY = e.clientY || (e.touches && e.touches[0].clientY);

    return { x: clientX - rect.left, y: clientY - rect.top };

  }

  function startDrawing(e) { e.preventDefault(); isDrawing = true; ctx.beginPath(); ctx.moveTo(getPos(e).x, getPos(e).y); ctx.lineWidth = 4; ctx.lineCap = "round"; ctx.strokeStyle = "#1e293b"; }

  function draw(e) { if (!isDrawing) return; e.preventDefault(); ctx.lineTo(getPos(e).x, getPos(e).y); ctx.stroke(); }

  function stopDrawing(e) { if(e) e.preventDefault(); isDrawing = false; }

  canvas.addEventListener('mousedown', startDrawing); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseup', stopDrawing); canvas.addEventListener('mouseout', stopDrawing);

  canvas.addEventListener('touchstart', startDrawing, {passive: false}); canvas.addEventListener('touchmove', draw, {passive: false}); canvas.addEventListener('touchend', stopDrawing);


  async function submitSignature() {

    const blank = document.createElement('canvas'); blank.width = canvas.width; blank.height = canvas.height;

    if (canvas.toDataURL() === blank.toDataURL()) return alert("⚠️ 請家長先在方框內簽名！");


    const base64Image = canvas.toDataURL('image/png');

    const btn = document.getElementById('submit-btn');

    btn.innerHTML = "⏳ 儲存中..."; btn.disabled = true;


    const res = await apiCall('save', { studentData: selectedStudent, base64Image: base64Image });

    alert(res.message);

    if(res.success) backToSearch();

    btn.innerHTML = "✅ 確認簽名送出"; btn.disabled = false;

  }


  // ================= 清單與報表 =================

  async function fetchRecentRecords() {

    document.getElementById('list-content').innerHTML = "<div style='padding:20px; text-align:center;'>載入中...</div>";

    const res = await apiCall('recent');

    if(res.success) {

      if (res.data.length === 0) return document.getElementById('list-content').innerHTML = "<div style='padding:20px; text-align:center;'>無紀錄。</div>";

      let html = `<table><tr><th>時間</th><th>班級</th><th>姓名</th><th>簽名</th></tr>`;

      res.data.forEach(r => { html += `<tr><td>${r.datetime}</td><td>${r.className}</td><td>${r.name}</td><td><img src="${r.imgUrl}" class="sign-img"></td></tr>`; });

      document.getElementById('list-content').innerHTML = html + `</table>`;

    }

  }


  document.getElementById('report-date').valueAsDate = new Date();

  async function fetchReport() {

    const targetDate = document.getElementById('report-date').value;

    if (!targetDate) return alert("請選擇日期");

    document.getElementById('report-content').innerHTML = "<div style='padding:20px; text-align:center;'>報表產生中...</div>";

    

    const res = await apiCall('report', { targetDate: targetDate });

    if(res.success) {

      if (res.data.length === 0) return document.getElementById('report-content').innerHTML = "<div style='padding:20px; text-align:center;'>這天沒有紀錄。</div>";

      let html = `<button class="btn btn-secondary" style="margin-bottom:15px;" onclick="window.print()">🖨️ 列印此頁</button>`;

      html += `<h2 style="text-align:center;">超時接回清冊 (${targetDate})</h2>`;

      html += `<div class="table-responsive"><table><tr><th>時間</th><th>班級</th><th>學號</th><th>姓名</th><th>簽名</th></tr>`;

      res.data.forEach(r => { html += `<tr><td>${r.time}</td><td>${r.className}</td><td>${r.studentId}</td><td>${r.name}</td><td><img src="${r.imgUrl}" class="sign-img"></td></tr>`; });

      document.getElementById('report-content').innerHTML = html + `</table></div>`;

    }

  }
